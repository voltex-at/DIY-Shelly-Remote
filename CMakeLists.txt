cmake_minimum_required(VERSION 3.16)

option(DSR_DIAGNOSTIC_BUILD "Build the unconfigured serial diagnostic target" OFF)
set(DSR_BOARD "" CACHE STRING "Hardware profile: lcdwiki-es3c35p")
if(DSR_DIAGNOSTIC_BUILD AND DSR_BOARD)
    message(FATAL_ERROR "Select a diagnostic target OR a hardware profile")
endif()
if(DSR_BOARD STREQUAL "lcdwiki-es3c35p")
    if(DEFINED IDF_TARGET AND NOT IDF_TARGET STREQUAL "esp32s3")
        message(FATAL_ERROR "LCDWiki ES3C35P requires IDF_TARGET=esp32s3")
    endif()
    set(IDF_TARGET "esp32s3" CACHE STRING "" FORCE)
    set(SDKCONFIG_DEFAULTS "sdkconfig.defaults;boards/lcdwiki-es3c35p/sdkconfig.defaults")
    add_compile_definitions(DSR_BOARD_ES3C35P=1)
elseif(NOT DSR_DIAGNOSTIC_BUILD)
    message(FATAL_ERROR "Select -DDSR_BOARD=lcdwiki-es3c35p or -DDSR_DIAGNOSTIC_BUILD=ON")
endif()
set(PROJECT_VER "0.7.56-dev")
# Release builds (.github/workflows/release.yml) pass the exact tag version.
if(DEFINED DSR_VERSION AND NOT DSR_VERSION STREQUAL "")
    set(PROJECT_VER "${DSR_VERSION}")
endif()
include($ENV{IDF_PATH}/tools/cmake/project.cmake)
project(diy_shelly_remote)
